
# 1. A tiny project with a local remote (a stand-in for a GitHub clone)

$ ls

# 2. Save this checkout's defaults (one review check: the tests)

$ pai init --default-agent claude --base main --check "test=python3 test_shop.py" --workers 2 --no-input | tail -4
  pai task new "What the agent should do" --body "How to check it" --write
  pai run --dry-run
  pai run
(pai config edit changes any default; project shop)
